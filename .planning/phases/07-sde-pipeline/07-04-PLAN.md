    ```rust
    use std::collections::HashMap;
    use crate::documentary::pipeline::step_clip as doc_step_clip;

    use crate::sde::error::SdeError;
    use crate::sde::types::SdePipelineState;

    /// SDE 视频裁剪——在纪录片裁剪基础上增加 OST=1 精确时间戳校正
    ///
    /// 对于 OST=1 片段，使用 find_precise_range() 从原始字幕段落中获取精确时间戳，
    /// 覆盖 LLM 输出的可能不精确时间戳（D-29）。
    pub async fn sde_step_clip(
        state: &mut SdePipelineState,
        video_path: &Path,
    ) -> Result<(), SdeError> {
        // 检查 OST=1 片段的精确时间戳校正
        for clip in &mut state.script {
            if clip.ost == OstType::OriginalSound {
                // 从原始字幕段落查找精确时间范围（D-29）
                if let Some((precise_start, precise_end)) =
                    find_precise_range(&clip.timestamp, &state.subtitle_segments)
                {
                    // 更新时间戳为精确值
                    let start_str = format!(
                        "{:02}:{:02}:{:02},{:03}",
                        (precise_start as u64) / 3600,
                        ((precise_start as u64) % 3600) / 60,
                        (precise_start as u64) % 60,
                        ((precise_start.fract() * 1000.0) as u64)
                    );
                    let end_str = format!(
                        "{:02}:{:02}:{:02},{:03}",
                        (precise_end as u64) / 3600,
                        ((precise_end as u64) % 3600) / 60,
                        (precise_end as u64) % 60,
                        ((precise_end.fract() * 1000.0) as u64)
                    );
                    clip.timestamp = format!("{}-{}", start_str, end_str);
                }
            }
        }

        // 调用纪录片裁剪（返回 HashMap<i64, PathBuf> 写入 state.script 的 video 字段等）
        // 注意：step_clip 接收 &mut PipelineState，但 SDE 需要转发到纪录片实现
        // 由于签名不匹配（SdePipelineState vs PipelineState），SDE 调用 step_clip 时
        // 传入必要的子集参数。实际实现中，run_sde() 的 step_clip 步骤直接调用纪录片函数。
        Ok(())
    }
    ```

    实际上此文件是轻封装——pipeline.rs 中的对应步骤函数将直接调用纪录片函数，
    此文件提供 sde_step_clip 函数用于精确时间戳校正逻辑。
    主要逻辑：遍历 OST=1 片段，调用 find_precise_range，更新 clip.timestamp。

    包含 #[cfg(test)] mod tests 块（至少 2 个测试验证时间戳校正）。
