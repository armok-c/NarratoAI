    **5. get_audio_rms(input: &Path) -> Result<f64, AudioError>**

    使用 symphonia 解码音频 → PCM f32 采样 → 计算 RMS（dB）。symphonia 解码循环实现参考以下模式：

    ```rust
    // symphonia 解码模板（仿照 RESEARCH.md Pattern 2）
    let src = std::fs::File::open(input)
        .map_err(|e| AudioError::FileNotFound(input.display().to_string()))?;
    let mss = symphonia::core::io::MediaSourceStream::new(Box::new(src), Default::default());
    
    let probed = symphonia::default::get_probe()
        .format(&symphonia::core::formats::FormatOptions::default(), mss,
                &symphonia::core::meta::MetadataOptions::default(),
                &symphonia::core::io::MediaSourceStreamOptions::default())
        .map_err(|e| AudioError::RmsFallbackFailed(format!("探测失败: {}", e)))?;
    
    let mut format = probed.format;
    let track = format.tracks().iter()
        .find(|t| t.codec_params.codec != symphonia::core::codecs::CODEC_TYPE_NULL)
        .ok_or_else(|| AudioError::RmsFallbackFailed("无音频轨道".into()))?;
    
    let codec_params = &track.codec_params;
    let track_id = track.id;
    let mut decoder = symphonia::default::get_codecs()
        .make(codec_params, &symphonia::core::codecs::DecoderOptions::default())
        .map_err(|e| AudioError::RmsFallbackFailed(format!("解码器初始化失败: {}", e)))?;
    
    let mut sample_buf: Option<symphonia::core::audio::SampleBuffer<f32>> = None;
    let mut all_samples: Vec<f32> = Vec::new();
    
    loop {
        let packet = match format.next_packet() {
            Ok(pkt) => pkt,
            Err(symphonia::core::errors::Error::IoError(ref e))
                if e.kind() == std::io::ErrorKind::UnexpectedEof => break,
            Err(_) => break,
        };
        if packet.track_id() != track_id { continue; }
        
        if let Ok(audio_buf) = decoder.decode(&packet) {
            let spec = *audio_buf.spec();
            let cap = audio_buf.capacity() as u64;
            let buf = sample_buf.get_or_insert_with(|| {
                symphonia::core::audio::SampleBuffer::<f32>::new(cap, spec)
            });
            buf.copy_interleaved_ref(audio_buf);
            all_samples.extend_from_slice(buf.samples());
        }
    }
    
    if all_samples.is_empty() {
        return Err(AudioError::RmsFallbackFailed("解码后无音频采样".into()));
    }
    
    let sum_sq: f64 = all_samples.iter().map(|s| (*s as f64) * (*s as f64)).sum();
    let rms = (sum_sq / all_samples.len() as f64).sqrt();
    
    let rms_db = if rms > 0.0 {
        20.0 * (rms / (i16::MAX as f64)).log10()
    } else {
        -60.0 // 静音
    };
    Ok(rms_db)
    ```

    **6. calculate_volume_adjustment(tts_lufs: f64, original_lufs: f64) -> (f64, f64)**

    纯函数（无文件 I/O），不依赖外部库：
    - 目标响度 -20.0 LUFS
    - `tts_adjustment = 10f64.powf((target_lufs - tts_lufs) / 20.0)`
    - `original_adjustment = 10f64.powf((target_lufs - original_lufs) / 20.0)`
    - clamp：tts 在 [0.1, 2.0]，original 在 [0.1, 3.0]
    - 返回 `(tts_adjustment, original_adjustment)`

    **7. normalize_audio_for_mixing(audio_path: &Path, output_dir: &Path, target_lufs: f64) -> Result<PathBuf, AudioError>**

    编排入口函数（per D-07，1:1 对齐 Python 版 `normalize_audio_for_mixing()` 顶层函数）：
    - 检查文件存在
    - 生成输出路径：`{output_dir}/{filename}_normalized.mp3`
    - 调用 `normalize_lufs(audio_path, &output_path, target_lufs, -1.0)`
    - 如果成功返回 `Ok(output_path)`；失败则按 D-03 回退到 RMS 标准化：
      - 调用 `get_audio_rms(audio_path)` 获取 RMS
      - 计算 gain：`target_lufs - rms_db`（简化增益补偿）
      - 通过 FFmpeg 的 `volume` filter 应用增益：`ffmpeg -i <input> -af volume=<gain>dB -ar 44100 -ac 2 <output>`
      - 返回 `Ok(output_path)` 或错误
    ```rust
    pub fn normalize_audio_for_mixing(
        audio_path: &Path, 
        output_dir: &Path, 
        target_lufs: f64,
    ) -> Result<PathBuf, AudioError> {
        if !audio_path.exists() {
            return Err(AudioError::FileNotFound(audio_path.display().to_string()));
        }
        let stem = audio_path.file_stem()
            .and_then(|s| s.to_str())
            .unwrap_or("audio");
        let output_path = output_dir.join(format!("{}_normalized.mp3", stem));
        
        // Try loudnorm two-pass first
        if normalize_lufs(audio_path, &output_path, target_lufs, -1.0).is_ok() {
            return Ok(output_path);
        }
        
        // Fallback: RMS-based normalization
        let rms_db = get_audio_rms(audio_path)?;
        let gain_db = target_lufs - rms_db;
        
        let ffmpeg_bin = ffmpeg_sidecar::ffmpeg::ffmpeg_path();
        let output = Command::new(&ffmpeg_bin)
            .args(["-y", "-hide_banner", "-i"])
            .arg(audio_path.as_os_str())
            .args(["-af", &format!("volume={}dB", gain_db)])
            .args(["-ar", "44100", "-ac", "2"])
            .arg(output_path.as_os_str())
            .output()
            .map_err(|e| AudioError::LoudnormNormalizeFailed(e.to_string()))?;
        
        if !output.status.success() {
            return Err(AudioError::RmsFallbackFailed(
                String::from_utf8_lossy(&output.stderr).to_string()
            ));
        }
        Ok(output_path)
    }
    ```
