    ```rust
    // 使用 tokio::process::Command 异步调用 yt-dlp
    async fn _get_video_formats(url: &str, proxy_url: Option<&str>) -> Result<Vec<VideoFormat>, YoutubeError> {
        // URL 前缀校验（防止参数注入）
        if !url.starts_with("http://") && !url.starts_with("https://") {
            return Err(YoutubeError::InvalidUrl(url.to_string()));
        }
        
        let mut cmd = Command::new("yt-dlp");
        cmd.args(["--dump-json", "--no-warnings", "--quiet", url]);
        
        if let Some(proxy) = proxy_url {
            cmd.args(["--proxy", proxy]);
        }
        
        let output = cmd.output().await
            .map_err(|e| YoutubeError::SubprocessError(e.to_string()))?;
        
        if !output.status.success() {
            let stderr = String::from_utf8_lossy(&output.stderr);
            return Err(YoutubeError::SubprocessError(stderr.to_string()));
        }
        
        let info: serde_json::Value = serde_json::from_slice(&output.stdout)
            .map_err(|e| YoutubeError::SubprocessError(e.to_string()))?;
        
        let formats = info["formats"].as_array()
            .ok_or_else(|| YoutubeError::FormatNotFound("无格式数组".into()))?;
        
        let result: Vec<VideoFormat> = formats.iter().map(|f| {
            let resolution = f["format_note"].as_str().unwrap_or("N/A").to_string();
            // 标准化分辨率： "1080p60" → "1080p"
            let base_res = if resolution.contains('p') {
                let parts: Vec<&str> = resolution.split('p').collect();
                format!("{}p", parts[0])
            } else {
                resolution.clone()
            };
            
            VideoFormat {
                format_id: f["format_id"].as_str().unwrap_or("").to_string(),
                resolution: base_res,
                ext: f["ext"].as_str().unwrap_or("").to_string(),
                vcodec: f["vcodec"].as_str().unwrap_or("none").to_string(),
            }
        }).collect();
        
        Ok(result)
    }
    ```

    **7. download_video(url: &str, resolution: &str, output_format: &str, rename: Option<&str>, proxy_url: Option<&str>) -> Result<DownloadResult, YoutubeError>**（per D-11, D-08, D-10）：

    ```rust
    pub async fn download_video(
        url: &str,
        resolution: &str,
        output_format: &str,
        rename: Option<&str>,
        proxy_url: Option<&str>,
    ) -> Result<DownloadResult, YoutubeError> {
        // 1. 校验 URL 格式
        if !url.starts_with("http://") && !url.starts_with("https://") {
            return Err(YoutubeError::InvalidUrl(url.to_string()));
        }
        
        // 2. 校验输出格式
        validate_format(output_format)?;
        
        // 3. 标准化分辨率： "1080p60" → "1080p"
        let base_resolution = if resolution.contains('p') {
            let parts: Vec<&str> = resolution.split('p').collect();
            format!("{}p", parts[0])
        } else {
            resolution.to_string()
        };
        
        // 4. 获取格式列表，匹配目标分辨率
        let formats = _get_video_formats(url, proxy_url).await?;
        
        // 5. 在匹配到目标分辨率的格式中，优先选 headerless/非 av01 的
        let matched = formats.iter()
            .filter(|f| f.resolution == base_resolution && f.vcodec != "none")
            .find(|f| !f.vcodec.contains("av01"));
        
        let format_id = match matched {
            Some(f) => f.format_id.clone(),
            None => {
                // fallback: 匹配任何有视频的格式
                let fallback = formats.iter()
                    .find(|f| f.vcodec != "none")
                    .map(|f| f.format_id.clone())
                    .ok_or_else(|| YoutubeError::FormatNotFound(
                        format!("无可用视频格式（目标分辨率: {}）", base_resolution)
                    ))?;
                fallback
            }
        };
        
        // 6. 下载（使用 tokio::process::Command，无进度回调 per D-10）
        let output_template = match rename {
            Some(name) => name.to_string(),
            None => format!("{}_video.%(ext)s", resolution),
        };
        
        let mut cmd = Command::new("yt-dlp");
        cmd.args([
            "-f", &format!("{}+bestaudio", format_id),
            "--merge-output-format", output_format,
            "--no-warnings",
            "--quiet",
            "-o", &output_template,
            url,
        ]);
        
        if let Some(proxy) = proxy_url {
            cmd.args(["--proxy", proxy]);
        }
        
        let output = cmd.output().await
            .map_err(|e| YoutubeError::SubprocessError(e.to_string()))?;
        
        if !output.status.success() {
            let stderr = String::from_utf8_lossy(&output.stderr);
            return Err(YoutubeError::DownloadFailed(stderr.to_string()));
        }
        
        // 7. 获取视频标题
        let title_info = Command::new("yt-dlp")
            .args(["--get-title", "--no-warnings", "--quiet", url])
            .output().await
            .ok();
        let title = title_info
            .filter(|o| o.status.success())
            .and_then(|o| String::from_utf8(o.stdout).ok())
            .map(|s| s.trim().to_string())
            .unwrap_or_else(|| "unknown_title".to_string());
        
        // 构建输出路径（yt-dlp 输出模板会在当前目录或指定路径）
        let output_path = PathBuf::from(rename.unwrap_or(&format!("{}_video", resolution)));
        let output_path = output_path.with_extension(output_format);
        
        Ok(DownloadResult { output_path, title })
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user-provided URL → yt-dlp subprocess | untrusted URL passed as CLI argument. CRITICAL |
| yt-dlp subprocess → downloaded file | untrusted file written to disk |
| proxy URL → yt-dlp --proxy | credential exposure risk |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-11-01 | Tampering | `download_video()` yt-dlp CLI | mitigate | URL prefix must be http/https. Use `Command::args()` with array — NEVER shell strings. URL passed as isolated arg. No `format!()` on URL strings. References: ASVS V5.1.1 (input validation) |
| T-11-02 | Spoofing | `_get_video_formats()` URL validation | mitigate | URL prefix check (`starts_with("http://")` or `starts_with("https://")`). Rejects `file://`, `ftp://`, `--config`, etc. References: ASVS V5.3.1 (input validation) |
| T-11-03 | Information Disclosure | proxy URL exposure | accept | Proxy URL from config.toml may contain credentials. Documented in PROJECT.md: proxy auth via env vars recommended, not config file. yt-dlp --proxy passes URL as-is. |
| T-11-04 | Denial of Service | very large download | accept | No file size limit enforced at this layer. Phase 6 or user workflow limits total download time. yt-dlp itself handles throttling and retry. |
</threat_model>
