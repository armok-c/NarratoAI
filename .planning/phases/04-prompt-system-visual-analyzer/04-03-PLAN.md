    ```rust
    pub mod documentary;
    pub mod short_drama_editing;
    pub mod short_drama_narration;

    use crate::prompt::error::PromptError;
    use crate::prompt::registry::PromptRegistry;

    /// 注册错误包装类型
    #[derive(Debug)]
    pub struct RegistrationErrors(pub Vec<PromptError>);

    impl std::fmt::Display for RegistrationErrors {
        fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
            write!(f, "Prompt 注册失败 ({} 个错误):", self.0.len())?;
            for (i, err) in self.0.iter().enumerate() {
                write!(f, "\n  [{}] {}", i + 1, err)?;
            }
            Ok(())
        }
    }

    impl std::error::Error for RegistrationErrors {}

    /// 注册所有内置 Prompt（D-17）
    pub fn register_all_prompts(registry: &mut PromptRegistry) -> Result<(), RegistrationErrors> {
        let mut errors = Vec::new();

        documentary::register_documentary_prompts(registry, &mut errors);
        short_drama_editing::register_short_drama_editing_prompts(registry, &mut errors);
        short_drama_narration::register_short_drama_narration_prompts(registry, &mut errors);

        if errors.is_empty() { Ok(()) } else { Err(RegistrationErrors(errors)) }
    }
    ```
