export function useTheme() {
  const appStore = useAppStore()
  const vuetifyTheme = useVuetifyTheme()
  vuetifyTheme.global.name.value = appStore.theme
  function toggleTheme() { appStore.toggleTheme(); vuetifyTheme.global.name.value = appStore.theme }
  function setTheme(val: 'light' | 'dark') { appStore.setTheme(val); vuetifyTheme.global.name.value = val }
