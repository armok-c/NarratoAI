       pub struct VideoSegment {
           material: VideoMaterial,
           speed: Speed,
           target_timerange: Timerange,
           source_timerange: Option<Timerange>,
       }

       impl VideoSegment {
           /// 创建视频片段——有处理后的视频文件（per D-09 有 video 时）
           pub fn new(path: &Path, target: Timerange, width: u32, height: u32) -> Result<Self, JianYingError> {
               Ok(Self {
                   material: VideoMaterial::new(path, target.duration, width, height)?,
                   speed: Speed::new(),
                   target_timerange: target,
                   source_timerange: None,  // 有处理后的文件，不需要 source_timerange
               })
           }

           /// 创建视频片段——回退到原始视频路径（per D-09 无 video 时）
           pub fn with_source_timerange(
               path: &Path,
               target: Timerange,
               source: Timerange,
               width: u32,
               height: u32,
           ) -> Result<Self, JianYingError> {
               Ok(Self {
                   material: VideoMaterial::new(path, target.duration, width, height)?,
                   speed: Speed::new(),
                   target_timerange: target,
                   source_timerange: Some(source),
               })
           }

           /// 导出为 VideoSegmentJson
           pub fn to_json(&self) -> VideoSegmentJson {
               let base = self.build_base_segment();
               // clip: ClipTransform::default_video()
               // uniform_scale: UniformScale { on: true, value: 1.0 }
               // hdr_settings: HdrSettings { intensity: 1.0, mode: 1, nits: 1000 }

           pub fn material_id(&self) -> &str { &self.material.material_id }
           pub fn speed_id(&self) -> &str { &self.speed.id }
       }
