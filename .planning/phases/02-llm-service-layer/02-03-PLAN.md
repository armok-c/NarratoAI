    **步骤 A — 创建 src/llm/register.rs**

    实现 `register_all_providers()` 函数（D-07）。从 `AppConfig` 读取配置，创建 `OpenAiCompatibleProvider` 实例，注册到 `Registry`。

    ```rust
    use std::sync::Arc;
    use crate::config::types::AppConfig;
    use crate::llm::registry::Registry;
    use crate::llm::openai_compatible::OpenAiCompatibleProvider;

    /// 注册所有活跃 provider（D-07）
    ///
    /// 在库初始化时调用。读取 AppConfig 的 vision/text 配置段和 proxy 配置，
    /// 创建 OpenAiCompatibleProvider 实例并注册到 Registry。
    ///
    /// 注册逻辑：
    /// - 若 vision_openai_api_key 非空 → 注册 "openai_vision" provider
    /// - 若 text_openai_api_key 非空 → 注册 "openai_text" provider
    /// - provider 名称与 D-08 的默认 provider 选择逻辑配合
    ///
    /// 代理支持（D-16）：proxy.enabled=true 时透传 proxy.http / proxy.https 到 provider 构造函数
    pub fn register_all_providers(config: &AppConfig, registry: &mut Registry) {
        let proxy_http = if config.proxy.enabled && !config.proxy.http.is_empty() {
            Some(config.proxy.http.clone())
        } else {
            None
        };
        let proxy_https = if config.proxy.enabled && !config.proxy.https.is_empty() {
            Some(config.proxy.https.clone())
        } else {
            None
        };

        // 注册 vision provider（D-15: timeout from config）
        if !config.app.vision_openai_api_key.is_empty()
            && !config.app.vision_openai_model_name.is_empty()
        {
            if let Ok(provider) = OpenAiCompatibleProvider::new(
                config.app.vision_openai_api_key.clone(),
                config.app.vision_openai_model_name.clone(),
                config.app.vision_openai_base_url.clone(),
                config.app.llm_max_retries,
                config.app.llm_vision_timeout,
                proxy_http.clone(),
                proxy_https.clone(),
            ) {
                registry.register("openai_vision", Arc::new(provider));
            } else {
                tracing::warn!("vision provider 注册失败（配置错误）");
            }
        }

        // 注册 text provider（D-15: timeout from config）
        if !config.app.text_openai_api_key.is_empty()
            && !config.app.text_openai_model_name.is_empty()
        {
            if let Ok(provider) = OpenAiCompatibleProvider::new(
                config.app.text_openai_api_key.clone(),
                config.app.text_openai_model_name.clone(),
                config.app.text_openai_base_url.clone(),
                config.app.llm_max_retries,
                config.app.llm_text_timeout,
                proxy_http,
                proxy_https,
            ) {
                registry.register("openai_text", Arc::new(provider));
            } else {
                tracing::warn!("text provider 注册失败（配置错误）");
            }
        }
    }
    ```

    注意事项：
    - 使用 `if let Ok(...)` 而非 `unwrap()` ——构造函数失败时不 panic，仅 warn 日志
    - `proxy_http` 和 `proxy_https` 在两个 provider 之间共享（第一个注册克隆 proxy_http，第二个 move）
    - `proxy_http.clone()` ——第一次注册 clone，第二次直接 move 剩余所有权
    - register 的 name 使用 "openai_vision" 和 "openai_text" 作为 key——与 D-08 的默认 provider 选择配合，调用方通过 `config.app.vision_llm_provider` 决定默认 provider 名称
